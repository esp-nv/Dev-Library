• Локален AI Сървър: Използваш Ollama – безплатен инструмент, който се инсталира на компютъра ти и ти позволява да въртиш мощни, но леки модели (като Llama 3.1 8B или Microsoft Phi-3) напълно офлайн.

3. История и Памет (Data Ентусиаст частта)

💻 РЕАЛНИЯТ КОД: Твоето Офлайн AI (Python)

Отвори VS Code, създай файл с име local_ai.py и постави този чист код. Той прави точно това, което поиска: има умен превключвател (Интернет vs Офлайн), чете локален файл (текст/код) и пази история в списък.
python
import os
import json
import urllib.request

# --- НАСТРОЙКА НА СИСТЕМАТА ---
USE_INTERNET = False  # Смени на True, ако искаш да ползваш уеб API
MODEL_NAME = "phi3:mini"
HISTORY_FILE = "chat_history.json"

# --- 1. УПРАВЛЕНИЕ НА ИСТОРИЯТА ---
def load_history():
    if os.path.exists(HISTORY_FILE):
        with open(HISTORY_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    return []

def save_history(history):
    with open(HISTORY_FILE, "w", encoding="utf-8") as f:
        json.dump(history, f, ensure_ascii=False, indent=4)

# --- 2. ЧЕТЕНЕ НА ЛОКАЛЕН ФАЙЛ (Твоят проект) ---
def read_local_file(file_path):
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            return f.read()
    except Exception as e:
        return f"Грешка при четене на файла: {e}"

# --- 3. ХИБРИДНО AI ЯДРО (Online / Offline Switch) ---
def ask_ai(prompt, system_context="", history=[]):
    # Сглобяваме контекста от историята + новия въпрос
    full_messages = []
    if system_context:
        full_messages.append({"role": "system", "content": system_context})
    
    for h in history[-5:]:  # Пазим само последните 5 реплики, за да не претоварим паметта
        full_messages.append(h)
        
    full_messages.append({"role": "user", "content": prompt})

    if not USE_INTERNET:
        # --- ИЗПЪЛНЕНИЕ НАПЪЛНО ОФЛАЙН ЧРЕЗ OLLAMA ---
        url = "http://localhost:11434/api/chat"
        data = {
            "model": MODEL_NAME,
            "messages": full_messages,
            "stream": False
        }
        
        try:
            req = urllib.request.Request(url, method="POST")
            req.add_header("Content-Type", "application/json")
            jsondata = json.dumps(data).encode("utf-8")
            
            with urllib.request.urlopen(req, data=jsondata) as response:
                result = json.loads(response.read().decode("utf-8"))
                return result["message"]["content"]
        except Exception:
            return "Грешка: Увери се, че Ollama е стартирана и моделът е свален!"
    else:
        # --- ТУК СЕ СЛАГА ОНЛАЙН API (напр. OpenAI, Claude или уеб линк) ---
        return "Автоматично превключване: Извиквам Онлайн Модел в облака..."

# --- 4. ГЛАВЕН СЪРЦЕВИДЕН ПРОЦЕС ---
if __name__ == "__main__":
    print(f"🤖 Твоето Персонално AI Стартира (Офлайн Режим: {not USE_INTERNET})")
    
    # Зареждаме старата история от диска
    chat_history = load_history()
    print(f"📁 Заредени {len(chat_history) // 2} предишни разговора.")

    # Пример за подаване на локален файл (може да е твоя Blueprint или .cs файл)
    # Създаваме тестов файл за демонстрация
    with open("student_blueprint.txt", "w", encoding="utf-8") as f:
        f.write("Проект: StudentManagement. Слой: Data. Елементи: List<Student>.")
        
    print("\n📝 AI анализира локален файл 'student_blueprint.txt'...")
    file_content = read_local_file("student_blueprint.txt")
    
    context = f"Ти си локален AI асистент. Ето съдържанието на текущия проект: {file_content}"
    user_question = "Каква структура от данни използва проектът според файла?"
    
    print(f"👤 Ти питаш: {user_question}")
    
    # Питаме AI
    ai_response = ask_ai(user_question, system_context=context, history=chat_history)
    print(f"\n🤖 AI Отговаря:\n{ai_response}\n")
    
    # Записваме новия разговор в историята
    chat_history.append({"role": "user", "content": user_question})
    chat_history.append({"role": "assistant", "content": ai_response})
    save_history(chat_history)
    print("💾 Разговорът е записан в chat_history.json!")
Use code with caution.
