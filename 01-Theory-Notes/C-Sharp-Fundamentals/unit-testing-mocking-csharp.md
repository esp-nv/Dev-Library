// Unit (Единица) тестването проверява най-малкото възможно изолирано

// Проблемът: В реалния проект (StudentManagement) бизнес услугите
// зависят от бази данни, файлови системи и външни мрежи. Ако базата

using System;
using System.Collections.Generic;

// --- РЕАЛНИЯТ БИЗНЕС МОДЕЛ И КОНТАКТ (ИНТЕРФЕЙС) ---
public class Student
{
    public int Id { get; set; }
    public string Name { get; set; }
    public double AverageGrade { get; set; }
}

public interface IStudentRepository
{
    Student GetById(int id);
    void Update(Student student);
}

// --- УСЛУГАТА, КОЯТО ЩЕ ТЕСТВАМЕ ПОД ЛУПА ---
public class ScholarshipService
{
    private readonly IStudentRepository _repository;

    //Dependency Injection - Класът зависи САМО от договора (интерфейса)
    public ScholarshipService(IStudentRepository repository)
    {
        _repository = repository;
    }

    public bool IsEligibleForScholarship(int studentId)
    {
        // 1. Дърпаме студента от външния склад през интерфейса
        var student = _repository.GetById(studentId);
        
        if (student == null)
            throw new ArgumentException("Студентът не съществува в системата!");

        // 2. Бизнес правило: Само студенти с успех над 5.50 имат право на стипендия
        if (student.AverageGrade >= 5.50)
        {
            return true;
        }

        return false;
    }
}

// 2. АНАТОМИЯ НА ЖИЗНЕНИЯ ЦИКЪЛ НА ТЕСТОВАТА РАМКА
// ============================================================
// Какво се случва в RAM паметта, когато стартираш тестовете?
//   1. Рефлексивният скенер на NUnit/xUnit прочита метаданните.
//   2. Открива методи, маркирани с атрибути като [Test].
//   3. За ВСЕКИ отделен тест енджинът създава ЧИСТО НОВА инстанция
//      на тестовия клас в Heap паметта (Пълна изолация на състоянието).
//   4. След приключване на теста, инстанцията се изхвърля за Garbage Collector-а.
// ============================================================

// ============================================================
// 3. СТРУКТУРАТА AAA (Arrange, Act, Assert) В ДЕЙСТВИЕ
// ============================================================
// Всеки професионален тест е разделен на 3 чисти логически зони.
// Подаваме фалшив обект на репозиторито, за да тестваме САМО логиката на услугата.
// ============================================================
// 4. РЕАЛЕН ТЕСТОВ КОД (Изолация чрез Moq Рамката)
// ============================================================
// Използваме NUnit и Moq библиотеките, за да тестваме нашия
// ScholarshipService без реална връзка с базата данни.

using NUnit.Framework;
using Moq;

[TestFixture] // Рефлексивен маркер: Казва на енджина, че това е тестов клас
public class ScholarshipServiceTests
{
    private Mock<IStudentRepository> _mockRepo;
    private ScholarshipService _service;

    [SetUp] // Изпълнява се автоматично ПРЕДИ всеки отделен тест (Чиста RAM памет)
    public void Setup()
    {
        // Arrange (Фаза 1: Обща подготовка)
        // Библиотеката Moq генерира динамичен proxy-клас в паметта, който симулира интерфейса
        _mockRepo = new Mock<IStudentRepository>();
        
        // Инжектираме фалшивото репозитори в реалната бизнес услуга
        _service = new ScholarshipService(_mockRepo.Object);
    }

    [Test] // Маркер за единичен самостоятелен тест
    public void IsEligibleForScholarship_HighGrade_ShouldReturnTrue()
    {
        // Arrange (Допълнителна подготовка за конкретния тест)
        int targetId = 1;
        var fakeStudent = new Student { Id = targetId, Name = "Мария", AverageGrade = 5.80 };
        
        // Обучаваме нашия Mock обект: "Когато те викнат с Id 1, върни фалшивата Мария"
        _mockRepo.Setup(r => r.GetById(targetId)).Returns(fakeStudent);

        // Act (Фаза 2: Извикваме реалния метод под лупа)
        bool result = _service.IsEligibleForScholarship(targetId);

        // Assert (Фаза 3: Сверяване на очакванията и резултата в Stack-а)
        Assert.That(result, Is.True, "Студент с оценка над 5.50 трябва да има право на стипендия!");
    }

    [Test]
    public void IsEligibleForScholarship_StudentNotFound_ShouldThrowArgumentException()
    {
        // Arrange: Конфигурираме Mock-а да върни 'null' (няма такъв студент)
        int invalidId = 999;
        _mockRepo.Setup(r => r.GetById(invalidId)).Returns((Student)null);

        // Act & Assert: Проверяваме дали кодът хвърля правилната грешка при празна референция
        Assert.Throws<ArgumentException>(() => _service.IsEligibleForScholarship(invalidId));
    }

    [Test]
    public void UpdateStudent_ShouldTriggerRepositorySave()
    {
        // --- ПОВЕДЕНЧЕСКИ ТЕСТ (Behavior Testing) ---
        // Искаме да проверим дали нашият метод не просто връща стойност, 
        // а реално извиква записващия метод на базата данни.
        
        int targetId = 2;
        var fakeStudent = new Student { Id = targetId, Name = "Петър", AverageGrade = 5.60 };
        _mockRepo.Setup(r => r.GetById(targetId)).Returns(fakeStudent);

        // Извикваме услугата
        _service.IsEligibleForScholarship(targetId);

        // Проверка под капака: Извикан ли е методът GetById точно 1 път?
        _mockRepo.Verify(r => r.GetById(targetId), Times.Once, "Репозиторито трябва да бъде извикано точно веднъж!");
    }
}
