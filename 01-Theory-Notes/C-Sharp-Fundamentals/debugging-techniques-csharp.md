//   3. Когато C# срещне червена точка (Breakpoint), той инжектира

// Нишката (Thread) замръзва моментално. Изпълнението спира, но

using System;
using System.Collections.Generic;

public class Student
{
    public int Id { get; set; }
    public string Name { get; set; }
    public List<double> Grades { get; set; } = new List<double>();
}

public class GradeService
{
    public double CalculateAverage(int studentId)
    {
        // 🟥 СЛОЖИ BREAKPOINT ТУК (F9)
        var student = FetchStudentFromStorage(studentId);
        
        // КРИТИЧЕН КАПАН: Ако колекцията Grades е празна, ще получим грешка!
        double sum = 0;
        foreach (var grade in student.Grades)
        {
            sum += grade;
        }

        return sum / student.Grades.Count; // ❌ Потенциално DivideByZeroException
    }

    private Student FetchStudentFromStorage(int id)
    {
        if (id == 99) return null; // ❌ Симулация на скрит бъг (NullReference)
        
        return new Student { Id = id, Name = "Димитър" };
    }
}

// 2. РАБОТА С CALL STACK (Хронологичното ДНК на грешката)
// ============================================================
// Call Stack (Стекът от извиквания) работи на принципа LIFO (Last In, First Out).
// Всяко извикване на метод заделя нова рамка (Stack Frame) в паметта.
//
// Симулация на Call Stack прозореца при краш:
// 
//  [Frame 3] -> GradeService.CalculateAverage(Int32 studentId) - НАЙ-ОТГОРЕ (Мястото на краша)
//  [Frame 2] -> Program.ProcessStudentData()
//  [Frame 1] -> Program.Main(String[] args) - НАЙ-ОТДОЛУ (Откъдето тръгна всичко)
//
// ЗАПОМНИ:
// Винаги чети Call Stack отгоре надолу, за да видиш точния метод, който е счупил паметта.
// ============================================================

public class ProgramExecutor
{
    public void Execute()
    {
        GradeService service = new GradeService();
        
        try
        {
            // Симулираме извикване, което ще хвърли NullReferenceException под капака
            service.CalculateAverage(99); 
        }
        catch (Exception ex)
        {
            // Рефлексията на дебъгера чете ex.StackTrace
            Console.WriteLine($"Грешка в стека: {ex.StackTrace}");
        }
    }
}

    public void LoopThroughStudents()

        for (int i = 1; i <= 5000; i++)

            // 🟥 BREAKPOINT ТУК с условие: student.Id == 2543
            // Процесорът ще прелети през първите 2542 итерации за милисекунди
            // и нишката ще замръзне точно на елемент 2543!

        }
    }

    }
}
